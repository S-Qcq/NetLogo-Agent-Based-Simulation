;;NOTA: il seguente codice è stato riadattato a partire dal codice esistente di bidding market per un esame universitario di tecniche di simulazione. Data l'inesperienza con il linguaggio di Netlogo, l'intelligenza artificiale ha giocato un ruolo importante nel cercare di tradurre le idee da esplore nel modello in linee di codice.

globals [
  min-price ;;prezzo minimo possibile (floor del mercato).
  population ;;numero totale di buyer e seller
  sales-per-tick ;;quante vendite avvengono mediamente per passo di simulazione
  starting-asking-price ;;valori iniziali dei prezzi di offerta
  starting-willing-to-pay ;;valori iniziali dei prezzi di domanda
  amount-high
  amount-low

  avg-per-buyer ;;statistiche per monitorare comportamento medio
  avg-per-seller ;;statistiche per monitorare comportamento medio
  total-sales ;;vendite totali cumulative
  remaining-supply ;;offerta residua ad un certo tick
  starting-money-actual ;;denaro iniziale effettivamente allocato agli agenti
  
  monopolist-seller ;;riferimento al venditore monopolista
  num-sellers ;;numero totale di venditori presenti
  
  ;;mode: scelta del tipo di mercato (monopolio, oligopolio, competitivo), è gestita dal widget chooser di NetLogo
]

;;Permette di dividere le turtles in categorie, viene indicato il nome collettivo degli agenti e il nome singolare dell'agente
breed [ shadows shadow ]
breed [ sellers seller ]
breed [ buyers buyer ]
breed [ pops pop ]

;; con -own dichiariamo le variabili proprie di una certa entità
;; Il significato di “own" è "appartiene a questo agente". Ogni turtle tiene separati i valori delle proprie variabili.

;; turtles-own dichiara variabili proprie di tutti i turtles, indipendentemente dal tipo (breed).
turtles-own [
  money
  next-xcor
  next-ycor
  percent
  my-shadow
]

;;variabili aggiuntive solo per i turtles del breed sellers.
sellers-own [
  items-for-sale
  asking-price
  starting-supply
  behavior-after-sale
  behavior-no-sale
  sold
  is-monopolist? ; nuovo attributo
]

;;variabili aggiuntive solo per i turtles del breed buyers.
buyers-own [
  want-to-buy
  willing-to-pay
  starting-demand
  behavior-after-purchase
  behavior-no-purchase
  bought
]




to setup ;; INIZIO DEL SETUP
  clear-all;; Pulisce tutte le turtles, patches, links e variabili globali.Serve per resettare la simulazione prima di avviarla.
  
  ;;set <variabile> <valore> assegna un valore iniziale a una variabile globale
  set min-price 0.01
  set population 10
  set total-sales 0
  set starting-asking-price 100
  set starting-willing-to-pay 10
  set amount-high 50
  set amount-low 25

  ; NOTA: mode può venire dal widget chooser nell'interfaccia
  ; Se il widget non esiste, usa un valore di default
  if not is-string? mode [ set mode "perfect-monopoly" ]
  
  ; determina numero venditori in base al tipo di mercato
  set num-sellers (ifelse-value
  mode = "perfect-monopoly" [1]
  mode = "oligopoly" [3]
  [population] ; in tutti gli altri casi più venditori
)

  
  ;==========================================
  ;      CREAZIONE DEI VENDITORI
  ;==========================================
create-ordered-sellers num-sellers [
  forward 8
  set my-shadow nobody ;;inizializza il riferimento al “shadow” a nulla
  set money 0 ;;i venditori iniziano senza denaro
  set is-monopolist? false ;;tutti i venditori partono come non-monopolisti, solo se nel widget chooser è selezionato monopolio allora diventa TRUE
  
  ;; --- assegna il monopolista se serve ---
  ;; Solo se il mercato è monopolistico (mode = "perfect-monopoly") e il turtle è il primo creato (who = 0, è come dire che il primo turtle è quello numerato come 0):
  if mode = "perfect-monopoly" and who = 0 [
    set is-monopolist? true ;;identifica il monopolista.
    set monopolist-seller self ;;identifica il monopolista.
    set color red ;;colore distintivo con cui viene mostrato il monopolista;;
    set items-for-sale (get-random-amount supply-distribution supply-amount) * 3.5 ;;il monopolista ha più merce disponibile, ci aiuta a vedere meglio l'evolversi nel tempo del prezzo se ha più merce da vendere
  ]
  
  ;; --- venditori normali ---
  ;; Sono tutti gli altri venditori che hanno quantità normale di merce (items-for-sale) generata casualmente.
  if not is-monopolist? [
    set items-for-sale get-random-amount supply-distribution supply-amount
    set color green ;; sono colorati di verde per distinguerli dal monopolista.
  ]
  
  set starting-supply items-for-sale ;;salva la quantità iniziale di merce (utile per statistiche o reset).
  set asking-price get-starting-value starting-asking-price ;;prezzo iniziale, eventualmente modificato da funzioni di randomizzazione
    
    ;; --- comportamento per tipo di mercato ---
    ;; COME AGISCE IL VENDITORE IN BASE ALLA 'MODE' SELEZIONATA NEL WIDGET
    
    ;; 1 MONOPOLIO
    ;; 2 OLIGOPOLIO
    ;; 3 MERCATO COMPETITIVO
    
    ;;1
    ifelse mode = "perfect-monopoly" [
      
      ;; monopolista unico
      set asking-price asking-price * 1.5 ;;Prezzo aumentato del 50%
      set behavior-after-sale [-> change-price 3.5] ;;Comportamento adattivo dopo la vendita: aumento del prezzo di 3.5 unità.
      set behavior-no-sale    [hide? -> if (not hide?) [change-price -0.5]];;Comportamento se non vende: diminuisce leggermente il prezzo (-0.5).
      
    ] [
      ;;2
      ifelse mode = "oligopoly" [
        ;; venditori in oligopolio
        set shape "triangle"
        set color green + 2
        set asking-price asking-price * 1.2 ;;Prezzo leggermente più alto (20%) ma non alto quanto il caso del monopolio perfetto, qui c'è competizione tra sole 3 grandi aziende
        set behavior-after-sale [-> change-price 3.0] ;;Dopo la vendita: aumento del prezzo +3.0.
        set behavior-no-sale    [hide? -> if (not hide?) [change-price -1.0]] ;;Se non vende: diminuzione più contenuta (-1.0).
        
      ] [
        
        ;;3
        ;; mercato competitivo (default)
        ;; definiamo le strategie di comportamento per il mercato competitivo con venditori “normal/desperate/mix”
        ;; In generale prezzo iniziale più basso o normale
        ;; Se nel modello (nell’interfaccia o in un’altra parte del codice) è stato scelto "mix of all", allora ogni venditore riceve casualmente un comportamento tra tre tipi:

        ;; 0 -> normal
        
        ;; 1 -> desperate
        
        ;; 2 -> random

        ;;Se invece seller-behavior non è "mix of all", allora mix-behavior = -1 e il tipo sarà determinato solo dal valore di seller-behavior (sceltro tramite widget chooser).
        
        let mix-behavior ifelse-value seller-behavior = "mix of all" [random 3] [-1]
        
        
        ;;Se il comportamento scelto è "normal" (da interfaccia) oppure il caso casuale mix-behavior = 0:
        ;;Dopo aver venduto, il venditore aumenta il prezzo di +2.5.
        ;;Se non vende, e non è nascosto (hide? serve a evitare che lo faccia se non visibile), abbassa il prezzo di -2.0.
        ifelse seller-behavior = "normal" or mix-behavior = 0 [
          set behavior-after-sale [-> change-price 2.5]
          set behavior-no-sale    [hide? -> if (not hide?) [change-price -2.0]]
        ] [
          
          ;; Se seller-behavior = "desperate" o mix-behavior = 1:
          ;; Dopo aver venduto, quasi non aumenta il prezzo (+0.7).
          ;; Se non vende, abbassa drasticamente il prezzo (-5.0).
          ifelse seller-behavior = "desperate" or mix-behavior = 1 [
            set behavior-after-sale [-> change-price 0.7]
            set behavior-no-sale    [hide? -> if (not hide?) [change-price -5.0]]
          ] [
            
            ;; Caso finale (cioè se mix-behavior = 2 oppure seller-behavior non è né "normal" né "desperate"):
            ;; Dopo ogni tick, i venditori modificano il prezzo casualmente.
            ;; random 11 - 5 produce un valore tra -5 e +5.
            set behavior-after-sale [-> change-price (random 11 - 5)]
            set behavior-no-sale    [-> change-price (random 11 - 5)]
          ]
        ]
        
      ]
    ]
  ]
 

  ;;==========================================
  ;;      CREAZIONE DEI COMPRATORI
  ;;==========================================
  ;;Crea un numero di buyers pari alla variabile globale population (che era impostata nel setup)
  create-ordered-buyers population [
    forward 13 ;;Sposta ogni buyer in avanti di 13 patch. Serve per distribuirli graficamente in modo distinto dai sellers (che si muovevano di 8 patch)
    facexy 0 0 ;;Fa sì che i buyers “guardino” verso il centro della scena (coordinate 0,0). Tipico accorgimento visivo nei modelli NetLogo di mercato: compratori e venditori si muovono l’uno verso l’altro.
    set my-shadow nobody ;;Inizializza il riferimento al “shadow” come nullo. Shadow rappresenta il fabbisogno dell'acquirente prima di diventare nascosto (smettere di acquistare)
    
    set want-to-buy get-random-amount demand-distribution demand-amount ;;quantità che il buyer desidera acquistare generata da get-random-amount, usando distribuzione e ammontare di domanda.
    set starting-demand want-to-buy ;;valore iniziale, è una copia di want-to-buy
    set money get-starting-value starting-money;;denaro iniziale disponibile calcolato da get-starting-value starting-money
    set willing-to-pay get-starting-value starting-willing-to-pay ;;prezzo massimo che il buyer è disposto a pagare per unità calcolato da get-starting-value starting-willing-to-pay
    ;;NOTA: get-random-amount e get-starting-value sono funzioni definite altrove nel codice.
    ;;NOTA: starting-money, starting-willing-to-pay, demand-distribution, demand-amount sono variabili globali impostate nel setup.
    
    
    ;; Come per i venditori il comportamento viene determinato nel widget chooser
    let mix-behavior ifelse-value buyer-behavior = "mix of all" [random 3] [-1]
    ;; Se vale "mix of all", allora mix-behavior diventa un numero casuale (0, 1 o 2), che seleziona un tipo di comportamento casuale per ogni buyer
    ;; Altrimenti vale -1, e il comportamento è uniforme per tutti i buyer (determinate da buyer-behavior).
    
    ;;COMPORTAMENTO NORMALE
    ifelse buyer-behavior = "normal" or mix-behavior = 0 [
      set behavior-after-purchase [-> change-payment -2.0 ];;Dopo aver acquistato, il buyer riduce la sua disponibilità a pagare (-2.0) → effetto “soddisfazione”: dopo l’acquisto, è meno disposto a spendere.
      set behavior-no-purchase    [-> change-payment  2.5 ];;Se non acquista, aumenta la sua disponibilità a pagare (+2.5) → effetto “frustrazione”: vuole comprare di più, è disposto a spendere di più.
      ;;NOTA: change-payment è una funzione (definita altrove) che modifica la variabile willing-to-pay
    ] [
      
      ;;COMPORTAMENTO DISPERATO
      ;; acquirenti molto motivati: vogliono assolutamente comprare e sono disposti a salire rapidamente di prezzo.
      ifelse buyer-behavior = "desperate" or mix-behavior = 1 [
        set behavior-after-purchase [-> change-payment -0.5 ] ;;Dopo aver comprato abbassa pochissimo la disponibilità a pagare (-0.5)
        set behavior-no-purchase    [-> change-payment  7.5 ] ;;Se non compra la aumenta fortemente (+7.5).
      ] [
        ;; COMPORTAMENTO RANDOM
        ;;Cambiamenti casuali, compresi tra -5 e +5. Simula acquirenti imprevedibili o mercati caotici.
        set behavior-after-purchase [-> change-payment (random 11 - 5)]
        set behavior-no-purchase    [-> change-payment (random 11 - 5)]
      ]
    ]
  ] 
  
  ;;=================================
  ;;   SHADOWS (componente grafica)
  ;;=================================
  ;; crea ombre per i venditori
  ;; create-ordered-shadows num-sellers crea un numero di shadows (un altro tipo di turtle) pari al numero di venditori (num-sellers).
  create-ordered-shadows num-sellers [
    set color 2 ;;imposta il colore (2 = verde chiaro nel sistema di NetLogo). Le ombre vengono così visualizzate in modo uniforme.
    ask one-of sellers with [my-shadow = nobody] [ set my-shadow myself ];;Ogni shadow viene collegata a un venditore come “ombra personale” (una specie di agente specchio). Ha un'utilità puramente visiva per arricchire il grafico. NON cambia il funzionamento
  ]
  
  ;; crea ombre per i compratori
  ;; Fa la stessa cosa, ma per i buyers. Crea tante ombre quanti sono i compratori (population). Ogni buyer riceve un’ombra assegnata.
  create-ordered-shadows population [
    set color 2
    ask one-of buyers with [my-shadow = nobody] [ set my-shadow myself ]
  ]
  
  ;;Calcola la domanda media per buyer e l’offerta media per seller. Queste medie verranno usate per:
  ;; -normalizzare i valori grafici (come la dimensione delle ombre);
  ;; -avere parametri di riferimento per grafici o statistiche.
  set avg-per-buyer (sum [starting-demand] of buyers) / (count buyers)
  set avg-per-seller (sum [starting-supply] of sellers) / (count sellers)

  ask sellers [ ;;per ogni venditore
    update-seller-display ;;aggiorna la grafica del venditore (colore, forma, etichetta, ecc.).
    let shadow-size 1 + (items-for-sale / avg-per-seller) ;;calcola la dimensione dell’ombra in base alla quantità di merce disponibile. Se il venditore ha più merce della media, la sua ombra è più grande.
    ;;Poi l’ombra (il turtle shadow) viene aggiornata
    ask my-shadow [
      move-to myself ;;posiziona l’ombra sul venditore.
      set heading [heading] of myself ;;la fa “guardare” nella stessa direzione del venditore.
      set size shadow-size ;;cambia la dimensione per rappresentare la quantità di offerta
      ;; NOTA: per il venditore le ombre servono come rappresentazione visiva della quantità di beni disponibili
    ]
  ]

  ask buyers [ update-buyer-display ] ;; Chiama la procedura update-buyer-display (definita altrove) per aggiornare la rappresentazione visiva dei compratori (colore, forma, grandezza, etichette, ecc.).

  set starting-money-actual sum [money] of buyers ;; Calcola la somma del denaro iniziale di tutti i buyers. 
                                                  ;;Memorizzata in starting-money-actual (una variabile globale). 
                                                  ;;Serve per confronti successivi (es. andamento della ricchezza totale nel tempo).

  reset-ticks ;; azzera il contatore dei tick (tempo della simulazione)
end ;; FINE DEL SETUP




;;==================================
;;           TO REPORT
;;==================================
;;to-report è l’equivalente di una funzione che ritorna un risultato in altri linguaggi di programmazione (come return in Python o function in JavaScript).


;;definisce una funzione con due parametri:
;;dist = tipo di distribuzione casuale (es. "even" o "exponential"),
;;amount = livello ("high" o "low").
to-report get-random-amount [ dist amount ]
  report ifelse-value dist = "even" [;; ifelse-value è come ifelse, ma usato dentro un’espressione: restituisce direttamente un valore.
    1 + random (get-amount amount);;genera un numero intero casuale uniforme tra 1 e get-amount amount
  ] [
    1 + floor random-exponential ((get-amount amount) / 2);;genera un numero secondo una distribuzione esponenziale, quindi con molti valori piccoli e pochi grandi. Il valore medio sarà circa (get-amount amount) / 2.
  ]
  ;;NOTA: cosa decide se il numero è generato da casuale uniforme o da una esponeziale?
  ;; Nella sezione setup ci sono:
  ;; set items-for-sale get-random-amount supply-distribution supply-amount
  ;; set want-to-buy get-random-amount demand-distribution demand-amount
  ;; quindi:
  ;; per i venditori (sellers): supply-distribution decide se le quantità offerte sono uniformi o esponenziali;
  ;; per i compratori (buyers): demand-distribution decide se le quantità domandate sono uniformi o esponenziali.
end

to-report get-amount [ amount ]
  report ifelse-value amount = "high" [ ;;amount è una stringa ("high" o "low").
    amount-high ;;Se è "high", restituisce il valore della variabile globale amount-high.
  ] [
    amount-low ;;Altrimenti, restituisce amount-low.
  ]
end

to-report get-starting-value [ starting-value ] ;;un valore medio iniziale
  report precision ((starting-value / 2) + random (starting-value / 2)) 2;;genera un numero casuale tra 0 e metà del valore iniziale 
                                                                         ;;quindi il valore finale è compreso tra metà e valore pieno (range: 0.5×starting-value → 1×starting-value).
end





to go ;; INIZIO GO
  ;; è la procedura principale che controlla l’evoluzione temporale del modello.
  ;; In NetLogo, ogni simulazione procede “a passi di tempo discreti” (i famosi tick).
  ;; fa muovere o aggiornare gli agenti,
  ;;fa avvenire interazioni (vendite, acquisti, crescita, ecc.),registra risultati,aggiorna la visualizzazione.
  
  ;;In pratica:
  ;;setup = crea il mondo (condizioni iniziali),
  ;;go = lo fa evolvere (dinamica del sistema).
  
  if (sum [items-for-sale] of sellers = 0 or (0 = count buyers with [money > 0 and want-to-buy > 0])) [ stop ];;Se non ci sono più beni da vendere (sum [items-for-sale] of sellers = 0)
                                                                                                              ;;oppure Se non esistono più compratori con soldi e voglia di comprare (count buyers with [money > 0 and want-to-buy > 0] = 0),
                                                                                                              ;;allora la simulazione si ferma con stop. Sta a significare che la simulazione si arresta quando il mercato è “esaurito”

  clear-drawing ;;cancella eventuali linee o segni tracciati nel mondo grafico (usato se gli agenti “disegnano” percorsi o connessioni).
  set sales-per-tick 0 ;;azzera il contatore delle vendite effettuate in questo tick.
  
  
  ;;======================================
  ;;SETTINGS PER IL MOVIMENTO CIRCOLARE
  ;;======================================
  ;;Calcolo delle posizioni “successive” per i compratori
  ;;Ogni buyer tiene in memoria dove si trova il buyer successivo, per spostarsi poi in quella posizione.
  ;;Serve  per creare un movimento circolare o un rimescolamento visivo dei compratori, così non restano fermi.
  ask buyers [
    let first-buyer min [who] of buyers ;;Calcola l’indice del primo (min [who]) 
    let last-buyer max [who] of buyers ;;e dell’ultimo (max [who]) buyer
    let next ifelse-value who = first-buyer [last-buyer] [who - 1]
    set next-xcor [xcor] of buyer next
    set next-ycor [ycor] of buyer next
  ]
  ;;ogni buyer si sposta  alle coordinate che aveva salvato nel passo precedente.
  ;;facexy 0 0 fa sì che ciascun buyer guardi verso il centro del mondo (coordinate 0,0).
  ask buyers [
    set xcor next-xcor
    set ycor next-ycor
    facexy 0 0
  ];;NOTA: E' un trucco visivo per mantenere i compratori ordinati o disposti in cerchio attorno al centro della scena.

  set remaining-supply (sum [items-for-sale] of sellers);;Calcola e memorizza la quantità totale di beni ancora in vendita in quel momento.
                                                        ;;Serve sia per le statistiche, sia per eventuali condizioni di terminazione o grafici.
                                                        ;;ticks mod population crea una rotazione che cambia l’ordine in cui i buyers vengono considerati.
                                                        ;;Aggiungendo 1, eviti lo zero e garantisci un indice valido.

  let offset 1 + ticks mod population;;offset viene usato per selezionare in modo ciclico i compratori che agiscono a ogni tick.
  
  
  ;;============================================================
  ;; LOGICA ECONOMICA ( chi compra da chi e in quale condizione)
  ;;=============================================================
  ;;Di seguito si trova la logica economica vera e propria del modello, ossia chi compra da chi e in quale scenario di mercato.
  ;;Questa parte di to go gestisce il commercio vero e proprio tra agenti, adattandolo al tipo di mercato scelto nel modello
  
  ; in monopolio, tutti i compratori interagiscono con il monopolista
  ifelse mode = "perfect-monopoly" [
  
  let the-seller one-of sellers with [is-monopolist?]
  ask buyers [ do-commerce-with the-seller ];;do-commerce-with è una procedura che gestisce la transazione: controlla prezzo, disponibilità, quantità acquistata, aggiornamento di soldi e beni, ecc.
  ;; sostanzialmente monopolio tutti i buyers comprano dallo stesso seller.
] [
      ; oligopolio o competitivo: distribuzione equa
      ;;Ordina tutti i venditori e compratori, per avere un ordine coerente e ripetibile.
      let all-sellers sort sellers
      let all-buyers sort buyers
      ;;Calcola quanti compratori deve servire ciascun venditore
      let buyers-per-seller ceiling (population / num-sellers)
      
      ;;Distribuisce i compratori tra i venditori, è un ciclo itera su ogni indice i (da 0 a num-sellers - 1).
      foreach (range 0 num-sellers) [ i ->
        let the-seller item i all-sellers
        ;;assegna il gruppo corretto di buyers a ogni seller: start-buyer = indice iniziale, end-buyer = indice finale (non oltre la popolazione totale)
        let start-buyer i * buyers-per-seller
        let end-buyer min (list ((i + 1) * buyers-per-seller) population)
        
        ;;sublist crea la “fetta” di compratori
        foreach (sublist all-buyers start-buyer end-buyer) [ the-buyer ->
          ask the-buyer [ do-commerce-with the-seller ];;Fa sì che ciascun buyer del gruppo commercia con il proprio seller
        
        ;;NOTA: Ogni venditore serve un gruppo di compratori diverso.
        ;; È con le linee di codice sopra che il modello rappresenta l’oligopolio o la concorrenza perfetta: i compratori sono suddivisi “equamente”.
        ]
      ]
    ]
  

 
  ;;Aggiornamento grafico e contatori
  ask buyers [update-buyer-display]
  ask sellers [update-seller-display]
  set total-sales (total-sales + sales-per-tick)
  ;;Aggiornamento delle “pop” (particelle decorative, aspetto puramente estetico)
  ask pops [
    ifelse (size < 0.1 or not stars?)
    [ die ]
    [
      set heading (atan (random-float 0.5 - 0.25) 1)
      jump 0.5
      set size size - 0.025
      set color color - 0.25
    ]
  ]
  ;;Controllo di coerenza
  ;;Nessun compratore deve avere la volontà di pagare (willing-to-pay) più di quanto possiede in realtà (money).
  ;;Se succede, viene lanciato un messaggio d’errore e la simulazione si blocca.
  if (any? buyers with [want-to-buy > 0 and willing-to-pay > money]) [ 
    error "Cannot have turtles that want to pay more than their cash!" 
  ]

  
  ;;IMPORTANTE!!!
  ;;Incrementa di 1 il “tempo di simulazione” (un passo).
  ;;È obbligatorio in ogni ciclo go per far funzionare grafici e dinamiche temporali.
  tick
end ;;FINE SEZIONE TO GO



;;COMPONENTI VISIVE DEL GRAFICO (update delle dimensioni e colore)
;;L’agente compratore mostra con il suo corpo quanto ha comprato e con la sua ombra quanto vuole ancora comprare.
;;shadow: segue la posizione del suo buyer, assume la stessa direzione (heading), mostra graficamente quanto il buyer vuole ancora comprare (grande = alta domanda residua).
to update-buyer-display
  if want-to-buy = 0 [
    set color 2
  ]
  set size 1 + (bought / avg-per-buyer)
  let shadow-size 1 + (want-to-buy / avg-per-buyer)
  ask my-shadow [
    move-to myself
    set heading [heading] of myself
    set size shadow-size
  ]
end
;;Il venditore mostra visivamente quanto gli resta da vendere (dimensione) e il suo ruolo nel mercato (colore).
to update-seller-display
  if items-for-sale = 0 [ set color 2 ]
  set size 1 + (items-for-sale / avg-per-seller)
  ; evidenzia il monopolista
  if is-monopolist? and items-for-sale > 0 [ set color red ]
end




;;Quando un compratore “fa commercio con” un venditore (do-commerce-with [the-seller]),
;;questa procedura controlla se la transazione può avvenire (condizioni di prezzo e disponibilità),
;;e se sì, aggiorna soldi, quantità, comportamento e visualizzazione di entrambi.
to do-commerce-with [ the-seller ]
  let asking [asking-price] of the-seller;;Legge il prezzo che il venditore (the-seller) sta chiedendo in quel momento.
                                         ;;Viene salvato nella variabile locale asking.
  
  ;;“Se il venditore ha ancora beni da vendere, il compratore vuole ancora comprare, ha abbastanza soldi, 
  ;; e il prezzo richiesto è minore o uguale a quanto è disposto a pagare
  ifelse ([items-for-sale] of the-seller > 0 and want-to-buy > 0 and asking <= money and asking <= willing-to-pay) [
    create-link the-seller self yellow ;;creazione di link giallo per “VENDITA AVVENUTA”
    set sales-per-tick (sales-per-tick + 1);;Incrementa il contatore delle vendite di questo tick
    set want-to-buy (want-to-buy - 1) ;;Il compratore riduce la quantità che desidera ancora acquistare.
    let price asking ;;Salva il prezzo effettivo pagato (uguale a asking del venditore).
    set money precision (money - price) 2 ;;Aggiorna il denaro del compratore: sottrae il prezzo. Arrotonda a 2 decimali
    set money ifelse-value money < min-price [0] [money] ;;Se dopo l’acquisto il denaro scende sotto la soglia minima (min-price), 
                                                         ;;lo forza a 0 (per evitare numeri negativi o errori di arrotondamento).
    set bought (bought + 1);;Incrementa il contatore di acquisti completati dal compratore.
    
    ask the-seller [;;aggiorna il venditore
      set items-for-sale (items-for-sale - 1);;Diminuisce la quantità in vendita
      set money precision (money + price) 2 ;;Aumenta i soldi di cassa (money)
      set sold (sold + 1) ;;incrementa il numero di vendite effettuate (sold)
      run behavior-after-sale ;;Esegui la strategia VENDITORE impostata dopo una vendita (behavior-after-sale) definita nel setup
    ]
    run behavior-after-purchase;;Fa eseguire al COMPRATORE la sua strategia dopo un acquisto definita nel setup
    create-star;;Crea una “particella decorativa” (tipo effetto grafico di una vendita riuscita)
  ] [
    ;;TENTATIVI DI ACQUISTO FALLITI
    create-link the-seller self blue;;indica visivamente un tentativo di commercio fallito.
    let hide? (sellers-ignore-full-buyers? and (want-to-buy = 0))
    ;;Se la variabile globale sellers-ignore-full-buyers? è vera (di solito un’impostazione del modello)
    ;;e il compratore non vuole comprare più nulla, allora hide? = vero.
    ask the-seller [ (run behavior-no-sale hide?) ]
    run behavior-no-purchase ;;Fa reagire il compratore a una mancata transazione
    
  ]
end

;;Crea un piccolo effetto grafico “stella” ogni volta che una transazione ha successo 
;;(viene chiamata dentro do-commerce-with).
to create-star
  if stars? [
    hatch-pops 1 [
      set color yellow
      set shape "star"
      set size 0.5
      set label ""
    ]
  ]
end

;;Disegna una linea colorata temporanea tra un venditore e 
;;un compratore per rappresentare un’interazione economica.
to create-link [ some-seller some-buyer some-color ]
  ask some-seller [
    let oc color
    let x xcor
    let y ycor
    set color some-color
    set pen-size 3
    pen-down
    move-to some-buyer
    pen-up
    setxy x y
    set color oc
  ]
end


;;Aggiorna il prezzo che un venditore chiede per i suoi beni (asking-price) 
;;in base a un cambiamento percentuale passato come argomento (change).
;;Viene richiamata all’interno dei comportamenti:
;;behavior-after-sale (se una vendita è andata a buon fine),
;;behavior-no-sale (se non è riuscita a vendere).
to change-price [ change ]
  let before asking-price;;Salva il prezzo precedente del venditore in una variabile locale before, per poter confrontare dopo se è cambiato.
  set percent 1 + (change / 100);;Converte il valore di change in un fattore moltiplicativo percentuale.
  set asking-price check-for-min-price (precision (percent * asking-price) 2);;Poi passa il risultato alla funzione check-for-min-price.mpedisce che il prezzo scenda sotto un minimo (min-price)
  if before = asking-price [;;Se, dopo il calcolo, il prezzo non è cambiato, esegue un controllo più fine.
    if change < 0 and before != min-price [
      set asking-price precision (asking-price - min-price) 2;;Se voleva diminuire (change < 0) e non ha ancora raggiunto il prezzo minimo,allora forza una piccola diminuzione assoluta pari a min-price.
    ]
    if change > 0 [
      set asking-price precision (asking-price + min-price) 2;;Analogamente, se voleva aumentare (change > 0) ma il prezzo non è cambiato (magari per arrotondamento),forza una piccola crescita assoluta di min-price.
    ]
  ]
end



;;l meccanismo con cui i compratori cambiano il loro “willingness to pay” (cioè la disponibilità a pagare).
;;Serve a rappresentare il comportamento dinamico della domanda.
to change-payment [ change ];;Salva la disponibilità a pagare prima della modifica, così da poter controllare se cambia davvero.
  let before willing-to-pay
  set percent 1 + (change / 100);;Converte la variazione in percentuale
  set willing-to-pay check-for-min-price (precision (percent * willing-to-pay) 2);;Aggiorna il valore in base alla percentuale e lo passa a check-for-min-price per assicurarsi 
                                                                                 ;;che non vada sotto il prezzo minimo (min-price).
  ;;se, dopo i calcoli, il valore non è cambiato (magari per arrotondamento o limite),
  ;;forza una piccola variazione assoluta di ± min-price
  if before = willing-to-pay [
    if change < 0 and before != min-price [
      set willing-to-pay precision (willing-to-pay - min-price) 2
    ]
    if change > 0 [
      set willing-to-pay precision (willing-to-pay + min-price) 2
    ]
  ]
  if willing-to-pay > money [ set willing-to-pay money ];;Nessuno può offrire più denaro di quanto possiede
end

to-report seller-cash;;Restituisce la somma totale di denaro posseduto da tutti i venditori.
  report sum [money] of sellers
end


;;Controlla se esiste almeno un venditore con is-monopolist? = true.
;;Se sì → restituisce il suo money (cioè il profitto/ricchezza del monopolista).
;;Se no → restituisce 0.
to-report monopolist-cash
  report ifelse-value any? sellers with [is-monopolist?] [
    [money] of one-of sellers with [is-monopolist?]
  ] [0]
end


;;Controlla se esiste almeno un venditore con is-monopolist? = true.
;;Se esiste → riporta il suo prezzo corrente (asking-price), arrotondato a due decimali.
;;Se non esiste (cioè non siamo in modalità monopolio) → restituisce 0.
;;in pratica: “Qual è il prezzo attuale del monopolista?”
to-report monopolist-price
  report ifelse-value any? sellers with [is-monopolist?] [
    precision [asking-price] of one-of sellers with [is-monopolist?] 2
  ] [0]
end


;;Seleziona tutti i venditori che non sono monopolisti → competitors.
;;Se ne esistono → calcola la media dei loro prezzi di vendita (asking-price), con due decimali.
;;Se non ci sono → riporta 0.
;; In pratica:“Qual è il prezzo medio dei venditori non monopolisti (competitori)?”
to-report competitors-avg-price
  let competitors sellers with [not is-monopolist?]
  report ifelse-value any? competitors [
    precision mean [asking-price] of competitors 2
  ] [0]
end


;;Controlla se ci sono state vendite totali (total-sales > 0) e se esiste un monopolista.
;;se si: Prende il numero di unità vendute dal monopolista, divide per vendite totali e moltiplica per 100
;;Se non ci sono vendite o monopolista → restituisce 0
to-report monopolist-market-share
  report ifelse-value total-sales > 0 and any? sellers with [is-monopolist?] [
    precision (100 * [sold] of one-of sellers with [is-monopolist?] / total-sales) 1
  ] [0]
end

;;Calcola la percentuale di beni venduti rispetto al totale iniziale disponibile:sold = beni già venduti. items-for-sale = beni ancora in vendita.
;;Che percentuale della merce totale è già stata venduta?
to-report percent-items-sold
  report precision (100 * sum [sold] of sellers / sum [items-for-sale + sold] of sellers) 1
end


;;“Quanta parte della domanda dei compratori è stata soddisfatta?”
to-report percent-demand-satisfied
  report precision (100 * sum [bought] of buyers / sum [want-to-buy + bought] of buyers) 1
end

;;È una funzione di sicurezza che impedisce che un prezzo scenda sotto il minimo (min-price).
to-report check-for-min-price [ value ]
  report precision ifelse-value value < min-price [min-price] [value] 2
end

; Reporter per analizzare il markup del monopolista
;;Recupera i prezzi medi
to-report monopolist-markup ;;Calcola il markup percentuale del monopolista rispetto alla media dei concorrenti:
  let mono-price monopolist-price
  let comp-price competitors-avg-price
  report ifelse-value comp-price > 0 [
    precision (100 * (mono-price - comp-price) / comp-price) 1
  ] [0]
end

; Reporter per vedere prezzo medio iniziale
to-report initial-avg-price
  report starting-asking-price * 0.75
end
